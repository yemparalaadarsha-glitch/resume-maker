\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & \textbf{\small #2} \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeProjectHeading}[2]{
    \item
    \begin{tabular*}{1.001\textwidth}{l@{\extracolsep{\fill}}r}
      \small#1 & \textbf{\small #2}\\
    \end{tabular*}\vspace{-7pt}
}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.0in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}

    \BLOCK{endfor}

 \begin{itemize}[leftmargin=0.15in, label={}]

 \end{itemize}
 \vspace{-16pt}

\section{Experience}
  \resumeSubHeadingListStart
\BLOCK{for job in experience}
    \resumeSubheading
      {\VAR{job.company}}{\VAR{job.start} -- \VAR{job.end}}
      {\VAR{job.title}}{\VAR{job.location}}
      \resumeItemListStart
\BLOCK{for bullet in job.bullets}
        \resumeItem{\VAR{bullet}}
\BLOCK{endfor}
      \resumeItemListEnd
\BLOCK{endfor}
  \resumeSubHeadingListEnd
\vspace{-16pt}

\BLOCK{if projects}
\section{Projects}
    \vspace{-5pt}
    \resumeSubHeadingListStart
\BLOCK{for project in projects}
      \resumeProjectHeading
          {\textbf{\VAR{project.name}}\BLOCK{if project.tech} $|$ \emph{\VAR{project.tech|join(', ')}}\BLOCK{endif}}{}
          \resumeItemListStart
\BLOCK{for bullet in project.bullets}
            \resumeItem{\VAR{bullet}}
\BLOCK{endfor}
          \resumeItemListEnd
          \vspace{-13pt}
\BLOCK{endfor}
    \resumeSubHeadingListEnd
\vspace{-1pt}
\BLOCK{endif}

\BLOCK{if certifications}
\section{Certifications}
    \resumeItemListStart
\BLOCK{for cert in certifications}
        \resumeItem{\BLOCK{if cert.credential_href}\href{\VAR{cert.credential_href}}{\VAR{cert.name}}\BLOCK{else}\VAR{cert.name}\BLOCK{if cert.credential_label} (\VAR{cert.credential_label})\BLOCK{endif}\BLOCK{endif}.}
\BLOCK{endfor}
    \resumeItemListEnd
\BLOCK{endif}
